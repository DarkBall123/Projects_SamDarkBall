In Arma 3:

setVehicleLock "UNLOCKED" = same as "lock 0"

setVehicleLock "DEFAULT" = same as "lock 1"

setVehicleLock "LOCKED" = same as "lock 2"

setVehicleLock "LOCKEDPLAYER" = same as "lock 3"
